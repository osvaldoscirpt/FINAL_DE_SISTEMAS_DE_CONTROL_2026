%% =========================================================
% SISTEMA RESIDUAL COMPENSADO
% Gc(s) = K' * (s+33.28)(s+228.57) / [ s(s+400) * ((s+23.97)^2 + 35.18^2) ]
% =========================================================

gain = 6.57;
Z1 = -228.57;
Z2 = -33.28;
p_real = 23.97;
p_imag = 35.18;
c = 400;

% Numerador y denominador
num = gain * poly([Z1 Z2]);
quad = [1, 2*p_real, p_real^2 + p_imag^2];
den  = conv([1 0], conv([1 c], quad));

Npad = [0 0 num];

% Polos y ceros
polos = [0; -c; -p_real+1j*p_imag; -p_real-1j*p_imag];
ceros = [Z1; Z2];

[~,ordP] = sort(real(polos)); polos = polos(ordP);
[~,ordZ] = sort(real(ceros)); ceros = ceros(ordZ);

n = length(polos);
m = length(ceros);

%% =========================================================
% MOSTRAR POLOS Y CEROS
% =========================================================
disp(' ')
disp('==========================================================')
disp(' POLOS DEL SISTEMA COMPENSADO')
disp('==========================================================')
for i = 1:n
    fprintf('Polo %d:  s = %10.5f %+.5fj\n', i, real(polos(i)), imag(polos(i)));
end

disp(' ')
disp('==========================================================')
disp(' CEROS DEL SISTEMA COMPENSADO')
disp('==========================================================')
for i = 1:m
    fprintf('Cero %d:   s = %10.5f %+.5fj\n', i, real(ceros(i)), imag(ceros(i)));
end

%% =========================================================
% PUNTOS DE ENTRADA / SALIDA
% =========================================================
dD = polyder(den);
dN = polyder(num);

A = conv(dD, num);
B = conv(den, dN);
maxL = max(length(A), length(B));
A = [zeros(1, maxL-length(A)) A];
B = [zeros(1, maxL-length(B)) B];
numBreak = A - B;

sBreak = roots(numBreak);

sReales = [];
KReales = [];
for i = 1:length(sBreak)
    s = sBreak(i);
    if abs(imag(s)) < 1e-6
        s = real(s);
        if abs(polyval(num, s)) > 1e-8
            K = -polyval(den, s) / polyval(num, s);
            sReales(end+1) = s;
            KReales(end+1) = K;
        end
    end
end
[sReales, ord] = sort(sReales);
KReales = KReales(ord);

disp(' ')
disp('==========================================================')
disp(' PUNTOS DE ENTRADA / SALIDA')
disp('==========================================================')
for i = 1:length(sReales)
    fprintf('Punto %d:  s = %10.4f   (K = %.4g)\n', i, sReales(i), KReales(i));
end

%% =========================================================
% CENTROIDE Y ASÍNTOTAS
% =========================================================
centroide = real( (sum(polos) - sum(ceros)) / (n-m) );
angulos = (2*(0:n-m-1)+1)*180/(n-m);

disp(' ')
disp('==========================================================')
fprintf(' CENTROIDE = %.4f\n', centroide)
fprintf(' Ángulos de asíntotas: %.1f° y %.1f°\n', angulos(1), angulos(2))
disp('==========================================================')

%% =========================================================
% RANGO DE K
% =========================================================
Kpos = logspace(-3, 5, 2500);
Kneg = -logspace(-3, 5, 2500);

%% =========================================================
% SEGUIMIENTO K > 0
% =========================================================
rootsPos = zeros(n, length(Kpos));
permsN = perms(1:n);

rActual = roots(den + Kpos(1)*Npad);
asignacion = zeros(1,n); usados = false(1,n);
for i = 1:n
    dist = abs(rActual - polos(i)); dist(usados) = inf;
    [~,idx] = min(dist); asignacion(i) = idx; usados(idx) = true;
end
rootsPos(:,1) = rActual(asignacion).';

for kidx = 2:length(Kpos)
    rActual = roots(den + Kpos(kidx)*Npad);
    rAnterior = rootsPos(:,kidx-1);
    mejorCosto = inf; mejorPerm = 1:n;
    for p = 1:size(permsN,1)
        candidato = rActual(permsN(p,:));
        costo = sum(abs(candidato(:) - rAnterior(:)));
        if costo < mejorCosto
            mejorCosto = costo; mejorPerm = permsN(p,:);
        end
    end
    rootsPos(:,kidx) = rActual(mejorPerm).';
end

%% =========================================================
% SEGUIMIENTO K < 0
% =========================================================
rootsNeg = zeros(n, length(Kneg));

rActual = roots(den + Kneg(1)*Npad);
asignacion = zeros(1,n); usados = false(1,n);
for i = 1:n
    dist = abs(rActual - polos(i)); dist(usados) = inf;
    [~,idx] = min(dist); asignacion(i) = idx; usados(idx) = true;
end
rootsNeg(:,1) = rActual(asignacion).';

for kidx = 2:length(Kneg)
    rActual = roots(den + Kneg(kidx)*Npad);
    rAnterior = rootsNeg(:,kidx-1);
    mejorCosto = inf; mejorPerm = 1:n;
    for p = 1:size(permsN,1)
        candidato = rActual(permsN(p,:));
        costo = sum(abs(candidato(:) - rAnterior(:)));
        if costo < mejorCosto
            mejorCosto = costo; mejorPerm = permsN(p,:);
        end
    end
    rootsNeg(:,kidx) = rActual(mejorPerm).';
end

%% =========================================================
% TRANSFORMACIÓN LOGARÍTMICA DEL EJE IMAGINARIO

transformarY = @(y) sign(y) .* log10(1 + abs(y));

yPosPlot = transformarY(imag(rootsPos));
