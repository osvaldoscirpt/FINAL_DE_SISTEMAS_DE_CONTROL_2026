%% =========================================================
% SISTEMA RESIDUAL COMPENSADO
% =========================================================

gain = 6.57;

Z1 = -228.57;
Z2 = -33.28;

p_real = 23.97;
p_imag = 35.18;

c = 228.57;                % <-- CAMBIÁ ESTE VALOR

% Tolerancia para detectar cancelaciones polo-cero
tolCancel = 1e-6;

%% =========================================================
% SISTEMA ORIGINAL
% =========================================================

% Numerador original
numOriginal = gain * poly([Z1 Z2]);

% Par de polos complejos
quad = [1, 2*p_real, p_real^2 + p_imag^2];

% Denominador original
denOriginal = conv([1 0], conv([1 c], quad));

%% =========================================================
% POLOS Y CEROS ORIGINALES
% =========================================================

polosOriginal = [ ...
     0;
    -c;
    -p_real + 1j*p_imag;
    -p_real - 1j*p_imag];

cerosOriginal = [ ...
    Z1;
    Z2];

% Ordenar polos
[~,ordP] = sort(real(polosOriginal));
polosOriginal = polosOriginal(ordP);

% Ordenar ceros
[~,ordZ] = sort(real(cerosOriginal));
cerosOriginal = cerosOriginal(ordZ);

%% =========================================================
% DETECCIÓN AUTOMÁTICA DE CANCELACIONES POLO-CERO
% =========================================================

polosTrabajo = polosOriginal;
cerosTrabajo = cerosOriginal;

canceladosPolos = [];
canceladosCeros = [];

i = 1;

while i <= length(polosTrabajo)

    if isempty(cerosTrabajo)
        break
    end

    % Distancia entre el polo actual y todos los ceros
    distancias = abs(cerosTrabajo - polosTrabajo(i));

    [distMin,idxMin] = min(distancias);

    % Si están suficientemente cerca => cancelación
    if distMin < tolCancel

        poloCancelado = polosTrabajo(i);
        ceroCancelado = cerosTrabajo(idxMin);

        canceladosPolos(end+1) = poloCancelado;
        canceladosCeros(end+1) = ceroCancelado;

        % Eliminar polo y cero
        polosTrabajo(i) = [];
        cerosTrabajo(idxMin) = [];

    else

        i = i + 1;

    end

end

%% =========================================================
% SISTEMA EFECTIVO
% =========================================================

polos = polosTrabajo;
ceros = cerosTrabajo;

n = length(polos);
m = length(ceros);

%% =========================================================
% CONSTRUCCIÓN AUTOMÁTICA DEL NUMERADOR REDUCIDO
% =========================================================

num = gain;

for i = 1:length(ceros)

    num = conv(num,[1 -ceros(i)]);

end

%% =========================================================
% CONSTRUCCIÓN AUTOMÁTICA DEL DENOMINADOR REDUCIDO
% =========================================================

den = 1;

for i = 1:length(polos)

    den = conv(den,[1 -polos(i)]);

end

% Eliminar pequeños errores numéricos
num(abs(num) < 1e-12) = 0;
den(abs(den) < 1e-12) = 0;

%% =========================================================
% PADDING DEL NUMERADOR
% =========================================================
% IMPORTANTE:
%
% Para poder calcular:
%
% den + K*num
%
% ambos vectores deben tener la misma cantidad de coeficientes.
%
% Por eso agregamos ceros a la izquierda del numerador.

Npad = [zeros(1,length(den)-length(num)) num];

%% =========================================================
% INFORMACIÓN DE CANCELACIONES
% =========================================================

disp(' ')
disp('==========================================================')
disp(' ANÁLISIS AUTOMÁTICO DE POLO-CERO')
disp('==========================================================')

if isempty(canceladosPolos)

    disp('No se detectaron cancelaciones polo-cero.')

else

    fprintf('Se detectaron %d cancelación(es):\n\n', ...
        length(canceladosPolos));

    for i = 1:length(canceladosPolos)

        fprintf('Cancelación %d:\n',i);

        fprintf('   Polo = %10.5f %+.5fj\n', ...
            real(canceladosPolos(i)), ...
            imag(canceladosPolos(i)));

        fprintf('   Cero = %10.5f %+.5fj\n\n', ...
            real(canceladosCeros(i)), ...
            imag(canceladosCeros(i)));

    end

end

fprintf('Polos originales : %d\n',length(polosOriginal));
fprintf('Ceros originales : %d\n',length(cerosOriginal));
fprintf('Cancelaciones    : %d\n',length(canceladosPolos));
fprintf('Polos efectivos  : %d\n',n);
fprintf('Ceros efectivos  : %d\n',m);
fprintf('RAMAS EFECTIVAS  : %d\n',n);

disp('==========================================================')

%% =========================================================
% PARÁMETROS RESPUESTA TEMPORAL
% =========================================================

Kp          = 50;
t_final     = 3;
y_min       = 0;
y_max       = 1.25;
dt          = 0.001;

%% =========================================================
% LÍMITE AUTOMÁTICO EJE REAL
% =========================================================

xMin = -(c + 50);
xMax = 80;
yMaxReal = 800;

%% =========================================================
% INFORMACIÓN EN CONSOLA
% =========================================================

disp(' ')
disp('==========================================================')
disp(' POLOS EFECTIVOS DEL SISTEMA')
disp('==========================================================')

for i = 1:n

    fprintf('Polo %d:  s = %10.5f %+.5fj\n', ...
        i,real(polos(i)),imag(polos(i)));

end

disp(' ')
disp('==========================================================')
disp(' CEROS EFECTIVOS DEL SISTEMA')
disp('==========================================================')

for i = 1:m

    fprintf('Cero %d:   s = %10.5f %+.5fj\n', ...
        i,real(ceros(i)),imag(ceros(i)));

end

%% =========================================================
% PUNTOS DE ENTRADA / SALIDA
% =========================================================

dD = polyder(den);
dN = polyder(num);

A = conv(dD,num);
B = conv(den,dN);

maxL = max(length(A),length(B));

A = [zeros(1,maxL-length(A)) A];
B = [zeros(1,maxL-length(B)) B];

numBreak = A-B;

sBreak = roots(numBreak);

sReales = [];
KReales = [];

for i = 1:length(sBreak)

    s = sBreak(i);

    if abs(imag(s)) < 1e-6

        s = real(s);

        if abs(polyval(num,s)) > 1e-8

            K = -polyval(den,s)/polyval(num,s);

            sReales(end+1) = s;
            KReales(end+1) = K;

        end

    end

end

[sReales,ord] = sort(sReales);
KReales = KReales(ord);

%% =========================================================
% INFORMACIÓN PUNTOS ENTRADA / SALIDA
% =========================================================

disp(' ')
disp('==========================================================')
disp(' PUNTOS DE ENTRADA / SALIDA')
disp('==========================================================')

if isempty(sReales)

    disp('No se encontraron puntos reales de entrada/salida.')

else

    for i = 1:length(sReales)

        fprintf('Punto %d:  s = %10.4f   (K = %.4g)\n', ...
            i,sReales(i),KReales(i));

    end

end

%% =========================================================
% CENTROIDE
% =========================================================

if n ~= m

    centroide = real((sum(polos)-sum(ceros))/(n-m));

else

    centroide = NaN;

end

disp(' ')
disp('==========================================================')

if ~isnan(centroide)

    fprintf(' CENTROIDE = %.4f\n',centroide);

else

    disp(' CENTROIDE = No definido');

end

disp('==========================================================')

%% =========================================================
% NÚMERO AUTOMÁTICO DE RAMAS
% =========================================================

numRamas = n;

fprintf('\nCantidad automática de ramas = %d\n',numRamas);

%% =========================================================
% SEGUIMIENTO DE RAMAS
% =========================================================

Kpos = logspace(-3,5,2500);
Kneg = -logspace(-3,5,2500);

% Permutaciones para asociar cada raíz con su rama
permsN = perms(1:n);

%% =========================================================
% K > 0
% =========================================================

rootsPos = zeros(n,length(Kpos));

% Primera posición
rActual = roots(den + Kpos(1)*Npad);

asignacion = zeros(1,n);
usados = false(1,n);

for i = 1:n

    dist = abs(rActual-polos(i));

    dist(usados) = inf;

    [~,idx] = min(dist);

    asignacion(i) = idx;

    usados(idx) = true;

end

rootsPos(:,1) = rActual(asignacion).';

% Seguimiento
for kidx = 2:length(Kpos)

    rActual = roots(den + Kpos(kidx)*Npad);

    rAnterior = rootsPos(:,kidx-1);

    mejorCosto = inf;
    mejorPerm = 1:n;

    for p = 1:size(permsN,1)

        candidato = rActual(permsN(p,:));

        costo = sum(abs(candidato(:)-rAnterior(:)));

        if costo < mejorCosto

            mejorCosto = costo;
            mejorPerm = permsN(p,:);

        end

    end

    rootsPos(:,kidx) = rActual(mejorPerm).';

end

%% =========================================================
% K < 0
% =========================================================

rootsNeg = zeros(n,length(Kneg));

% Primera posición
rActual = roots(den + Kneg(1)*Npad);

asignacion = zeros(1,n);
usados = false(1,n);

for i = 1:n

    dist = abs(rActual-polos(i));

    dist(usados) = inf;

    [~,idx] = min(dist);

    asignacion(i) = idx;

    usados(idx) = true;

end

rootsNeg(:,1) = rActual(asignacion).';

% Seguimiento
for kidx = 2:length(Kneg)

    rActual = roots(den + Kneg(kidx)*Npad);

    rAnterior = rootsNeg(:,kidx-1);

    mejorCosto = inf;
    mejorPerm = 1:n;

    for p = 1:size(permsN,1)

        candidato = rActual(permsN(p,:));

        costo = sum(abs(candidato(:)-rAnterior(:)));

        if costo < mejorCosto

            mejorCosto = costo;
            mejorPerm = permsN(p,:);

        end

    end

    rootsNeg(:,kidx) = rActual(mejorPerm).';

end

%% =========================================================
% TRANSFORMACIÓN LOGARÍTMICA
% =========================================================

transformarY = @(y) sign(y).*log10(1+abs(y));

yPosPlot = transformarY(imag(rootsPos));
yNegPlot = transformarY(imag(rootsNeg));

%% =========================================================
% COLORES AUTOMÁTICOS
% =========================================================

colores = lines(max(n,1));

%% =========================================================
% FIGURA
% =========================================================

figure( ...
    'Color','w', ...
    'Position',[40 40 1300 920])

%% =========================================================
% SUBPLOT 1 – LUGAR DE LAS RAÍCES
% =========================================================

subplot(2,1,1)

hold on
grid on
box on

%% =========================================================
% RAMAS K > 0 Y K < 0
% =========================================================

hPos = gobjects(n,1);
hNeg = gobjects(n,1);

for i = 1:n

    hPos(i) = plot( ...
        real(rootsPos(i,:)), ...
        yPosPlot(i,:), ...
        '-', ...
        'Color',colores(i,:), ...
        'LineWidth',2.2);

    hNeg(i) = plot( ...
        real(rootsNeg(i,:)), ...
        yNegPlot(i,:), ...
        '--', ...
        'Color',colores(i,:), ...
        'LineWidth',2.7);

end

%% =========================================================

hPolos = plot( ...

    'kx', ...
    'MarkerSize',12, ...
    'LineWidth',2.5);

% CENTROIDE Y ASÍNTOTA

    hCentroide = plot( ...

    hAsym = plot( ...

        'm--', ...
        'LineWidth',1.5);

end

%% =========================================================
% ETIQUETAS K > 0 / K < 0
% =========================================================

for i = 1:n

    xp = real(polos(i));
    yp = imag(polos(i));

    yp_plot = transformarY(yp);

    text( ...
        xp+10, ...

    text( ...
        xp-10, ...
        yp_plot-0.08*sign(yp+1e-6), ...
        'K < 0', ...
        'Color',colores(i,:), ...
        'FontSize',9, ...
        'FontWeight','bold', ...
        'BackgroundColor','w', ...
        'Margin',1, ...
        'HorizontalAlignment','right');

end

%% =========================================================
% EJES
% =========================================================

xline(0,'k-','LineWidth',1.1);
yline(0,'k-','LineWidth',1.1);

%% =========================================================
% TICKS EJE Y
% =========================================================

yTicksOriginal = [ ...
    -800 -600 -400 -200 -100 -60 -30 ...
     0 ...
     30 60 100 200 400 600 800];

yTickLabels = { ...
    '-800','-600','-400','-200','-100','-60','-30',...
    '0',...
    '30','60','100','200','400','600','800'};

yTicksPlot = transformarY(yTicksOriginal);

[yTicksPlot,ordT] = sort(yTicksPlot);

%% =========================================================

% =========================================================

end

handlesLegend(end+1) = hPolos;
